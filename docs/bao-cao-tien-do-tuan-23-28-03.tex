\section{Chi tiết công việc}
% ═══════════════════════════════════════════════════════

\subsection{Admin Dashboard -- Kiến trúc}

\textbf{Quyết định thiết kế:} Xây dựng web app riêng biệt thay vì tích hợp vào Zalo Mini App, vì:
\begin{itemize}[leftmargin=1.5em]
  \item Phòng Đào Tạo làm việc trên máy tính (desktop-first UX)
  \item Cần bảng dữ liệu lớn, import file, export báo cáo -- không phù hợp mobile
  \item Tách biệt về auth (Firebase Auth thay vì Zalo OAuth)
  \item Không ảnh hưởng bundle size của Zalo Mini App
\end{itemize}

\textbf{Tech stack:}
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Framework:} Vite 5 + React 18 + TypeScript 5.9 (consistent với Zalo app)
  \item \textbf{UI Library:} Ant Design 5 -- Vietnamese locale, Table với sort/filter/pagination
  \item \textbf{Charts:} Recharts (PieChart, BarChart cho dashboard)
  \item \textbf{Import/Export:} xlsx (SheetJS) -- hỗ trợ cả .xlsx và .csv
  \item \textbf{Deploy:} Firebase Hosting (cùng project, domain riêng)
\end{itemize}

\subsection{Các trang đã xây dựng (7 pages)}

\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{LoginPage:} Đăng nhập email/password, icon InHUST, kiểm tra role admin
  \item \textbf{DashboardPage:} Thống kê tổng quan (users, classes, sessions, pending requests) + PieChart (attendance distribution) + BarChart (user breakdown)
  \item \textbf{UsersPage:} Bảng người dùng với search, filter theo role, export Excel
  \item \textbf{ClassesPage:} CRUD lớp học + import lớp từ Excel + tải file mẫu
  \item \textbf{ClassDetailPage:} Chi tiết lớp, thêm SV thủ công, import SV từ Excel với preview validation, danh sách phiên điểm danh
  \item \textbf{AttendancePage:} Báo cáo điểm danh theo khoảng thời gian + filter theo lớp + export Excel
  \item \textbf{AbsenceRequestsPage:} Duyệt/từ chối đơn xin phép nghỉ với ghi chú
  \item \textbf{FraudReportsPage:} Xem báo cáo gian lận, expandable pattern details
\end{enumerate}

\subsection{Admin Authentication}

\begin{itemize}[leftmargin=1.5em]
  \item Sử dụng Firebase Authentication (email/password) -- tách biệt với Zalo OAuth
  \item Thêm role \texttt{"admin"} vào \texttt{UserRole} type
  \item Hook \texttt{useAdminAuth}: onAuthStateChanged $\rightarrow$ check Firestore role $\rightarrow$ auto sign-out nếu không phải admin
  \item \texttt{AdminGuard} component bảo vệ tất cả routes
  \item Tạo account đầu tiên qua Firebase REST API (seed script)
\end{itemize}

\subsection{Import/Export Excel}

\textbf{Import lớp học:}
\begin{itemize}[leftmargin=1.5em]
  \item Upload file $\rightarrow$ parse bằng xlsx $\rightarrow$ hiển thị preview table (valid/invalid) $\rightarrow$ xác nhận
  \item Cột bắt buộc: Tên lớp, Mã lớp. Tùy chọn: Giảng viên, Email GV, Face, Peer
  \item Tự động match GV theo tên hoặc email
\end{itemize}

\textbf{Import sinh viên:}
\begin{itemize}[leftmargin=1.5em]
  \item Upload file $\rightarrow$ parse $\rightarrow$ preview $\rightarrow$ xác nhận
  \item Tự động tạo UserDoc nếu SV chưa có trong hệ thống (dùng MSSV làm doc ID)
  \item Cột bắt buộc: MSSV, Họ tên. Tùy chọn: Email, Khoa
\end{itemize}

\textbf{File mẫu:} Nút ``Tải file mẫu'' ngay trên trang, tạo file .xlsx với dữ liệu ví dụ.

\subsection{Admin Cloud Functions}

\begin{itemize}[leftmargin=1.5em]
  \item \texttt{requireAdmin} middleware: verify Firebase Auth + check role === "admin" từ Firestore
  \item \texttt{adminDashboardStats}: aggregate thống kê từ nhiều collections
  \item \texttt{adminBulkAssignStudents}: import SV hàng loạt, tạo user doc nếu cần, batch write
  \item \texttt{adminReviewAbsenceRequest}: duyệt/từ chối đơn với validation
  \item \texttt{adminCreateAccount}: tạo admin account mới (chỉ admin hiện tại gọi được)
\end{itemize}

\subsection{Firestore Security Rules}

\begin{itemize}[leftmargin=1.5em]
  \item Thêm helper function \texttt{isAdmin()}
  \item Admin có quyền \texttt{read} tất cả users, \texttt{create/update} users và classes
  \item Admin có quyền \texttt{read} fraud reports
  \item Collection mới \texttt{absence\_requests}: SV create, GV/Admin read + update
\end{itemize}

\subsection{Deploy}

\begin{itemize}[leftmargin=1.5em]
  \item Tạo Firebase Hosting site riêng: \texttt{inhust-admin}
  \item URL: \url{https://inhust-admin.web.app}
  \item Deploy bằng: \texttt{firebase deploy --only hosting:admin}
  \item Vite build output tối ưu với code splitting (vendor, antd, firebase, charts)
\end{itemize}
