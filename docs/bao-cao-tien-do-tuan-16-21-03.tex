\section{Chi tiết công việc}
% ═══════════════════════════════════════════════════════

\subsection{Feature 1: Các bước điểm danh optional}

\textbf{Bối cảnh:} Hiện tại flow điểm danh gồm 4 bước cứng (QR $\rightarrow$ Face $\rightarrow$ Peer $\rightarrow$ Done). Giảng viên đề xuất cho phép tùy chọn bật/tắt bước Face Verification và Peer Exchange.

\textbf{Giải pháp đã triển khai:}
\begin{itemize}[leftmargin=1.5em]
  \item Thêm \texttt{faceRequired}, \texttt{peerRequired} vào \texttt{ClassDoc} và \texttt{SessionDoc}
  \item Cloud Function \texttt{startSession} đọc config từ ClassDoc $\rightarrow$ ghi vào SessionDoc (server-side source of truth)
  \item Cập nhật trust score logic trên cả CF (\texttt{scanPeer}, \texttt{submitFaceResult}, \texttt{calculateTrustScores}) và client (\texttt{computeTrustScore})
  \item \textbf{Teacher UI:} Thêm section ``Cài đặt điểm danh'' trong TeacherClassDetail với 2 toggle switch
  \item \textbf{Student UI:} StepIndicatorBar dynamic (2--4 bước tùy config), skip logic, adaptive done screen
\end{itemize}

\textbf{Files đã sửa:} 8 files (types, services, CF, hooks, pages)

\subsection{Feature 2: Điểm danh thủ công cho GV}

\textbf{Bối cảnh:} GV cần có khả năng điểm danh thủ công cho SV không thể check-in (hết pin, quên điện thoại...), với lý do và audit log.

\textbf{Giải pháp đã triển khai:}
\begin{itemize}[leftmargin=1.5em]
  \item Thêm \texttt{manualBy}, \texttt{manualReason}, \texttt{manualAt} vào \texttt{AttendanceDoc}
  \item Cloud Function \texttt{manualAttendance}: validate teacher permission, require reason, tạo/cập nhật attendance record
  \item Client service \texttt{manualCheckIn()} gọi CF
  \item \textbf{TeacherReview:} Modal nhập lý do, hiển thị badge ``Có mặt (GV)'' + lý do, nút hủy điểm danh
  \item \textbf{TeacherMonitor:} Nút ``Điểm danh thủ công'' $\rightarrow$ student picker $\rightarrow$ reason modal (cho phiên đang active)
  \item CSV export thêm cột ``Lý do thủ công''
\end{itemize}

\textbf{Files đã sửa:} 6 files (types, CF, index, service, 2 pages)
