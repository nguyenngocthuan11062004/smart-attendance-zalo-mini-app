\section{Chi tiết công việc}
% ═══════════════════════════════════════════════════════

\subsection{Face Register -- Viết lại hoàn toàn}

\textbf{Vấn đề cũ:} Module cũ yêu cầu chụp CCCD (mặt trước + mặt sau) + selfie, gọi Cloud Function để OCR và so khớp khuôn mặt qua eKYC API. Không cần thiết cho điểm danh.

\textbf{Giải pháp mới:}
\begin{itemize}[leftmargin=1.5em]
  \item Chỉ cần 2 ảnh selfie (chính diện + nghiêng 15°)
  \item Sử dụng \textbf{face-api.js} (@vladmandic/face-api) -- chạy hoàn toàn trên thiết bị
  \item Models: TinyFaceDetector (190KB) + FaceLandmark68Tiny (75KB) + FaceRecognition (6.1MB)
  \item Load từ CDN (jsdelivr), lazy import -- chỉ tải khi user vào Face Register
  \item Phát hiện khuôn mặt $\rightarrow$ tính descriptor 128 chiều $\rightarrow$ so khớp 2 ảnh $\rightarrow$ lưu descriptor vào Firestore
  \item Xác minh khi điểm danh: so sánh descriptor mới với descriptor đã lưu (Euclidean distance $< 0.6$)
  \item Thêm màn hình \textbf{đồng ý thu thập dữ liệu} (consent screen) trước khi chụp
\end{itemize}

\subsection{Chuyển Cloud Functions sang Firestore trực tiếp}

\textbf{Lý do:} Firebase project đang ở gói Spark (miễn phí), không deploy được Cloud Functions (cần Blaze plan + thẻ tín dụng).

\textbf{Services đã chuyển:}
\begin{itemize}[leftmargin=1.5em]
  \item \texttt{checkInStudent} -- tạo attendance record trực tiếp
  \item \texttt{addBidirectionalPeerVerification} -- cập nhật peerVerifications hai chiều
  \item \texttt{updateFaceVerification, teacherOverride, manualCheckIn} -- ghi Firestore
  \item \texttt{calculateTrustScores} -- tính client-side bằng \texttt{computeTrustScore()}
  \item \texttt{requestTeacherRole} -- ghi role trực tiếp (GV do admin cấp)
  \item \texttt{registerFace, verifyFace} -- face-api.js client-side
  \item \texttt{aiChat} -- gọi Groq API trực tiếp từ client
\end{itemize}

\textbf{Firestore Rules:} Tạm thời mở (allow read/write: true) để app hoạt động. Sẽ thêm Firebase Auth khi upgrade Blaze plan.

\subsection{Tuân thủ chính sách Zalo Mini App}

\textbf{Lần submit đầu:} Bị từ chối do vi phạm chính sách 6.1 -- hiển thị popup xin quyền userInfo ngay khi mở app.

\textbf{Các sửa đổi:}
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{6.1 -- Ngữ cảnh xin quyền:} Bỏ \texttt{authorize()} khỏi \texttt{signIn()}. Chỉ gọi \texttt{getUserID()} khi mở app (không popup). Xin quyền userInfo khi user nhấn ``Tiếp tục'' ở trang Login
  \item \textbf{6.3 -- Không bắt buộc quyền:} User từ chối vẫn dùng app với tên ``Zalo User''
  \item \textbf{localStorage $\rightarrow$ Zalo Storage:} Tạo wrapper \texttt{utils/storage.ts} dùng \texttt{setStorage/getStorage} từ ZMP SDK + localStorage fallback
  \item \textbf{Tách authorize scopes:} Chỉ xin \texttt{scope.userInfo} khi login, \texttt{scope.userPhonenumber} riêng khi cần
  \item \textbf{Ẩn /dev page:} \texttt{import.meta.env.DEV} -- chỉ hiện trong development
  \item \textbf{Ẩn Microsoft OAuth:} Redirect ngoài Zalo $\rightarrow$ vi phạm chính sách
  \item \textbf{Bundle $< 10$MB:} Xóa models từ public/ (dùng CDN), nén ảnh, xóa splash\_inhust.png $\rightarrow$ 3.3MB
\end{itemize}
